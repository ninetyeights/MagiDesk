cmake_minimum_required(VERSION 3.24)
project(MagiDeskDesktopDrawProbe LANGUAGES CXX)
if(NOT WIN32 OR NOT MSVC OR NOT CMAKE_SIZEOF_VOID_P EQUAL 8)
  message(FATAL_ERROR "This experimental probe requires MSVC and a 64-bit Windows target.")
endif()
add_library(MagiDeskDesktopDrawProbe SHARED DesktopDrawProbe.cpp)
target_compile_features(MagiDeskDesktopDrawProbe PRIVATE cxx_std_17)
target_compile_definitions(MagiDeskDesktopDrawProbe PRIVATE UNICODE _UNICODE WIN32_LEAN_AND_MEAN NOMINMAX)
target_compile_options(MagiDeskDesktopDrawProbe PRIVATE /W4 /WX /guard:cf)
target_link_options(MagiDeskDesktopDrawProbe PRIVATE /guard:cf /DYNAMICBASE /NXCOMPAT /SECTION:.mdprobe,RWS)
target_link_libraries(MagiDeskDesktopDrawProbe PRIVATE user32 comctl32 ole32 oleaut32 shell32 uuid)
